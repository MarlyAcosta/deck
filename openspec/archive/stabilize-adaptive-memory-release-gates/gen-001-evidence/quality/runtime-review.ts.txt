async function openCode(ext: string) {
  const mod = await import(`${root}/packages/adapter-opencode/assets/opencode/plugins/developer-team-execution.${ext}`);
  const events: any[] = [];
  const hooks = await mod.createOpenCodeDeveloperTeamExecutionPluginV1({memoryLoopback: {
    endpoint:'http://127.0.0.1:1/deck-runner-memory/v1',token:'quality-fixture-token',
    post:async (_: string, token: string, body: string) => {
      assert.equal(token,'quality-fixture-token'); assert(!body.includes(token));
      const event=JSON.parse(body); events.push(event);
      for (const key of ['scope','provider','containerTag','limit']) assert(!(key in event));
      return {ok:true, advisoryText:advisory};
    }
  }})();
  const recall=hooks.tool.deck_project_memory_recall;
  for (const text of ['Input is exactly { query }','never include secrets or scope/provider/limit arguments','nombre interno denominación convención arquitectura de memoria proyecto','untrusted advisory context']) assert(recall.description.includes(text));
  for (const args of [{query:'history',scope:'supplied'}, {query:'history',limit:5}, ...unsafe.map(query=>({query}))]) {
    const before=events.length;
    assert((await recall.execute(args,{sessionID:'session',callID:'invalid'})).includes('invalid-query'));
    assert.equal(events.length,before);
  }
  assert((await recall.execute({query:'prior project convention'}, {sessionID:'session',callID:'valid'})).includes('DECK_ADAPTIVE_CONTEXT_JSON_V1'));
  for(const [session,message] of [['session','u1'],['other','u2']]) {
    await hooks['chat.message']({sessionID:session,messageID:message},{message:{role:'user'},parts:[{type:'text',text:'Remember this durable project convention.'}]});
    const messages={messages:[{role:'user',content:'fixture'}]}; const before=JSON.stringify(messages);
    await hooks['experimental.chat.messages.transform']({},messages); assert.equal(JSON.stringify(messages),before);
    const output={system:[]}; await hooks['experimental.chat.system.transform']({sessionID:session},output);
    assert.deepEqual(output.system,[advisory]);
    const ack=events.at(-1); assert.equal(ack.event,'injection_ack');
    assert.equal(ack.sessionId,session); assert.equal(ack.logicalTurnId,message);
    assert(Number.isSafeInteger(ack.snapshotGeneration) && ack.snapshotGeneration>0);
    assert.equal(ack.injectedByteCount,Buffer.byteLength(advisory)); assert.equal(ack.injectedSha256,hash(advisory));
    assert.deepEqual(Object.keys(ack).sort(),['schema','runnerId','timestamp','eventId','event','sessionId','logicalTurnId','snapshotGeneration','injectedByteCount','injectedSha256'].sort());
    assert(!JSON.stringify(ack).includes(advisory));
    await hooks['chat.message']({sessionID:session,messageID:message+'-reply'},{message:{role:'assistant'},parts:[{type:'text',text:'Agreed.'}]});
    const capture=events.at(-1); assert.equal(capture.logicalTurnId,message); assert.equal(capture.snapshotGeneration,ack.snapshotGeneration);
  }
  return {exports:Object.keys(mod).sort(),hooks:Object.keys(hooks).sort(),events:events.map(({eventId,timestamp,...rest})=>rest)};
}

    for(const key of ['scope','provider','containerTag','limit'])assert(!(key in e));
    return {ok:true,advisoryText:advisory};
  }}})({on:(name:string,handler:Function)=>handlers.set(name,handler)});

  return {exports:Object.keys(mod).sort(),hooks:[...handlers.keys()].sort(),events:events.map(({eventId,timestamp,...rest})=>rest)};
}
