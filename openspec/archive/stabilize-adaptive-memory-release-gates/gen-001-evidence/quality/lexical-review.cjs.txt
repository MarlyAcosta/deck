const hash=b=>crypto.createHash('sha256').update(b).digest('hex');
const findings=[];
for(const adapter of ['opencode','pi']) {
  const a=fs.readFileSync(`${base}/bun-1.3.12-release-workflow/${adapter}/unminified.js`,'utf8');
  const b=fs.readFileSync(`${base}/bun-1.4.0-local/${adapter}/unminified.js`,'utf8');

  const sb=ts.createSourceFile('b.js',b,ts.ScriptTarget.Latest,true,ts.ScriptKind.JS);
  assert.equal(sa.parseDiagnostics.length,0);assert.equal(sb.parseDiagnostics.length,0);
  let count=0;
  function visit(n) {
    if(ts.isCallExpression(n)&&n.expression.getText(sa)==='__commonJS'&&ts.isArrowFunction(n.arguments[0])) {
      count++;const hazards=[];
      function lexical(k) {
        if(ts.isFunctionLike(k)&&!ts.isArrowFunction(k)) {
          if(k.name&&ts.isComputedPropertyName(k.name))lexical(k.name.expression);
          return;
        }
        if(ts.isClassDeclaration(k)||ts.isClassExpression(k)) {
          for(const h of k.heritageClauses||[])lexical(h);
          for(const m of k.members)if(m.name&&ts.isComputedPropertyName(m.name))lexical(m.name.expression);
          return;
        }
        if(k.kind===ts.SyntaxKind.ThisKeyword||k.kind===ts.SyntaxKind.SuperKeyword||ts.isMetaProperty(k)||(ts.isIdentifier(k)&&k.text==='arguments')||(ts.isCallExpression(k)&&k.expression.getText(sa)==='eval')) hazards.push({line:sa.getLineAndCharacterOfPosition(k.pos).line+1,kind:ts.SyntaxKind[k.kind]});

      }
      lexical(n.arguments[0].body);assert.deepEqual(hazards,[]);
    }
    ts.forEachChild(n,visit);
  }
  visit(sa);assert.equal(count,71);

  assert.equal(normalize(a),normalize(b));
  const folder=adapter==='opencode'?'opencode/plugins':'pi/extensions';
  const adopted=fs.readFileSync(`${root}/packages/adapter-${adapter}/assets/${folder}/developer-team-execution.generated.js`,'utf8').split('\n').slice(2).join('\n');
  const minified=fs.readFileSync(`${base}/bun-1.3.12-release-workflow/${adapter}/minified.js`,'utf8');
